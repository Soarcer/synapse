```javascript
// Add this constant near the top, after AGPL_HEADER:
const MIT_HEADER = `// SPDX-License-Identifier: MIT
// Copyright (c) 2026 Bernhard
`;

// Add this function:
function replaceAgplHeader(content) {
  // Replace AGPL SPDX identifier line
  const agplPattern = /\/\/ SPDX-License-Identifier: AGPL-3\.0-only\n/;
  const agplOrLaterPattern = /\/\/ SPDX-License-Identifier: AGPL-3\.0-or-later\n/;
  // Replace ee enterprise header too
  const eePattern =
    /\/\/ SPDX-License-Identifier: LicenseRef-Synapse-Enterprise\n\/\/ Copyright \(c\) 2026 Synapse Contributors\n\/\/ See \/ee\/LICENSE for terms\n/;

  let result = content
    .replace(agplPattern, '// SPDX-License-Identifier: MIT\n')
    .replace(agplOrLaterPattern, '// SPDX-License-Identifier: MIT\n')
    .replace(eePattern, MIT_HEADER)
    .replace(
      /\/\/ Copyright \(c\) 2026 Synapse Contributors\n/g,
      '// Copyright (c) 2026 Bernhard\n',
    );

  return result;
}
```

```typescript
// SPDX-License-Identifier: MIT
// Copyright (c) 2026 Bernhard
import * as path from 'path';
import Mocha from 'mocha';
import { glob } from 'glob';

export async function run(): Promise<void> {
  const mocha = new Mocha({ ui: 'tdd', color: true, timeout: 30000 });
  const testsRoot = path.resolve(__dirname, '.');
  const files = await glob('**/*.test.js', { cwd: testsRoot });
  for (const f of files) {
    mocha.addFile(path.resolve(testsRoot, f));
  }
  return new Promise((resolve, reject) => {
    mocha.run((failures) => {
      if (failures > 0) reject(new Error(`${failures} tests failed.`));
      else resolve();
    });
  });
}
```

```typescript
#!/usr/bin/env node
// SPDX-License-Identifier: MIT
// Copyright (c) 2026 Bernhard
/**
 * Stress test: floods the Synapse server with events and monitors memory.
 * Run: npx tsx scripts/stress_test.ts
 * Server must be running: pnpm --filter @synapse-ai/server dev
 */

const SERVER_URL = 'http://localhost:4000/api/events';
const EVENTS_PER_SECOND = 50;
const DURATION_SECONDS = 60;
const SAMPLE_INTERVAL_MS = 5000;

async function sendEvent(sessionId: string, i: number) {
  const event = {
    hook_event_name: 'PreToolUse',
    session_id: sessionId,
    tool_name: `stress_tool_${i % 10}`,
    tool_input: { query: `stress test event ${i}` },
    model: 'claude-opus-4-6',
    cwd: '/stress/test',
  };
  await fetch(SERVER_URL, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(event),
  });
}

async function checkMemory(): Promise<{ heapUsed: number; heapTotal: number }> {
  const res = await fetch('http://localhost:4000/api/health');
  const data = (await res.json()) as { memoryUsage?: { heapUsed: number; heapTotal: number } };
  return data.memoryUsage ?? { heapUsed: 0, heapTotal: 0 };
}

async function main() {
  const sessionId = `stress-${Date.now()}`;
  console.log(`Starting stress test: ${EVENTS_PER_SECOND} events/sec for ${DURATION_SECONDS}s`);
  console.log('');

  const samples: { time: number; heapUsed: number }[] = [];
  let eventCount = 0;
  const startTime = Date.now();

  // Sample memory periodically
  const sampler = setInterval(async () => {
    const mem = await checkMemory();
    const elapsed = (Date.now() - startTime) / 1000;
    samples.push({ time: elapsed, heapUsed: mem.heapUsed });
    const heapMB = (mem.heapUsed / 1024 / 1024).toFixed(1);
    console.log(`t=${elapsed.toFixed(0)}s | events=${eventCount} | heap=${heapMB}MB`);
  }, SAMPLE_INTERVAL_MS);

  // Flood with events
  const floodInterval = setInterval(async () => {
    const elapsed = Date.now() - startTime;
    if (elapsed > DURATION_SECONDS * 1000) {
      clearInterval(floodInterval);
      clearInterval(sampler);

      // Analysis
      console.log('\n--- Results ---');
      if (samples.length >= 2) {
        const firstHeap = samples[0].heapUsed;
        const lastHeap = samples[samples.length - 1].heapUsed;
        const growthPct = (((lastHeap - firstHeap) / firstHeap) * 100).toFixed(1);
        const growing = lastHeap > firstHeap * 1.2;
        console.log(`Heap growth: ${growthPct}% (${growing ? 'FAIL — >20% growth' : 'PASS'})`);
        console.log(`Total events sent: ${eventCount}`);
      }
      process.exit(0);
    }

    // Send batch
    const batch = Array.from({ length: Math.floor(EVENTS_PER_SECOND / 10) }, (_, i) =>
      sendEvent(sessionId, eventCount + i),
    );
    await Promise.allSettled(batch);
    eventCount += batch.length;
  }, 100);
}

main().catch(console.error);
```

_Findings from plan review — not blocking v0.1.0, address post-launch._

- **Version downgrade commit message**: `@synapse-ai/cli` goes from 0.9.0 → 0.1.0. Extend the commit message to explain: `"chore: normalize all packages to 0.1.0 (pre-publication; prior versions were internal milestones)"` so future git log readers aren't confused.

- **README test count is known**: The actual test count is **591** (verified 2026-03-11 in worktree). Update Chunk 4 Task 6 Step 6 to say "Change '2,000+ tests' (or whatever the README claims) to '591 tests'" rather than "verify and note."

- **VS Code Marketplace category `Machine Learning`**: The spec mandates `["Machine Learning", "Visualization", "Debuggers"]`. "Machine Learning" may attract ML model-builders who won't find Synapse relevant; `"Developer Tools"` is more accurate. Revisit category selection before or after first Marketplace submission based on discoverability data.

- **First-run experience for CLI**: If no Claude Code session is active, dashboard shows empty state. Consider auto-running `simulate_session.ts` on first launch so new users see something immediately. Deferred to v0.1.1.

- **Launch strategy beyond code**: Publishing to registries is not sufficient for recognition. Plan separately: Hacker News post, r/ClaudeAI, dev.to article, short X/Twitter thread. This is outside code scope.

- **Changelog tooling**: For ongoing releases, consider `changesets` or conventional-commits tooling. For v0.1.0, manual release notes are sufficient.

- **DCO enforcement**: CONTRIBUTING.md requires DCO sign-off but there's no automated GitHub check. Add `probot/dco` GitHub Action post-launch.

- **Stress test: `better-sqlite3` → WS leak verification**: If the WebSocket leak check in Task 14 Step 5 finds no `clients.delete(ws)` call, the fix is a single-line change to the server's `ws.on('close')` handler. Low risk but worth verifying.
